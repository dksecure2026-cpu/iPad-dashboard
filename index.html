function locateUser() { if (!navigator.geolocation) { alert("Geolocation is not supported by your browser. Please tap the map to place a pin."); return; } const coordsDisplay = document.getElementById('coords-display'); coordsDisplay.innerText = "Locating your position..."; navigator.geolocation.getCurrentPosition( pos => { const lat = pos.coords.latitude; const lng = pos.coords.longitude; recceMap.setView([lat, lng], 16); setPin(lat, lng); }, err => { // Friendly message under the map instead of a blocking popup if (err.code === 1) { // PERMISSION_DENIED coordsDisplay.innerText = "⚠️ GPS access blocked. Tap the map or paste coordinates below."; } else if (err.code === 2) { // POSITION_UNAVAILABLE coordsDisplay.innerText = "⚠️ GPS signal unavailable. Tap the map to drop a pin."; } else { coordsDisplay.innerText = "⚠️ GPS timed out. Tap the map to drop a pin."; } }, { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 } ); }
